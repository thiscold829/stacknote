/** * StackNote Mark 1.1 Backend - Core Logic * 대상 시트명: progress_manage */ function doPost(e) { var lock = LockService.getScriptLock(); try { // 동시성 제어 (데이터 충돌 방지) lock.waitLock(10000); var params = JSON.parse(e.parameter.payload); var action = params.action; var sheetName = params.sheetName || "progress_manage"; var data = params.projects; if (action === "write") { return writeToSheet(sheetName, data); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok: false, error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function writeToSheet(sheetName, data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ok: false, error: "Sheet not found"})) .setMimeType(ContentService.MimeType.JSON); } // 1. 헤더 정의 (순서가 매우 중요합니다) var headers = ["id", "full_name", "team", "pct", "status", "sheet_url", "order"]; // 2. 데이터 가공 (Object Array -> 2D Array) var rows = data.map(function(p) { return [ p.id || "", p.full_name || "", p.team || "", p.pct || 0, p.status || "idle", p.sheet_url || "", p.order || 0 ]; }); // 3. 시트 초기화 및 쓰기 (헤더 제외한 데이터 영역만 삭제 후 쓰기) if (sheet.getLastRow() > 1) { sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).clearContent(); } if (rows.length > 0) { sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); } return ContentService.createTextOutput(JSON.stringify({ok: true})) .setMimeType(ContentService.MimeType.JSON); } // GET 요청 테스트용 (필요 시) function doGet(e) { return ContentService.createTextOutput("StackNote API is Active."); }
